The workspace has no application files to patch, but this is the drop-in replacement:

```ts
type Label = "spam" | "abusive" | "fine";

type Classification = {
  label: Label | "review";
  probabilities: Record<Label, number>;
};

export async function classifyReview(review: string): Promise<Classification> {
  const response = await fetch("https://openrouter.ai/api/alpha/decisions", {
    method: "POST",
    headers: {
      Authorization: `Bearer ${process.env.OPENROUTER_API_KEY}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      // Pin this to the canonical model build you have probed.
      model: process.env.DECISION_MODEL ?? "typesafe/jev-1.13-20260917",
      state: { review },
      questions: {
        moderation: {
          type: "choice",
          instructions:
            "Which single category best describes this product review?",
          criteria: {
            spam: "Promotional, repetitive, deceptive, autogenerated, or irrelevant content.",
            abusive:
              "Insults, threats, harassment, hateful content, or targeted personal attacks.",
            fine: "A relevant review that is neither spam nor abusive.",
          },
        },
      },
    }),
  });

  if (!response.ok) {
    throw new Error(`Decision request failed: ${response.status}`);
  }

  const body = await response.json();
  const answer = body.answers?.moderation;

  if (answer?.type !== "choice" || !answer.probabilities) {
    throw new Error("Invalid moderation decision response");
  }

  const probabilities = answer.probabilities as Record<Label, number>;
  const label = (Object.keys(probabilities) as Label[]).reduce((best, key) =>
    probabilities[key] > probabilities[best] ? key : best
  );

  // Tune this from representative probes; ambiguous reviews go to review.
  const MIN_CONFIDENCE = 0.8;

  return {
    label: probabilities[label] >= MIN_CONFIDENCE ? label : "review",
    probabilities,
  };
}
```

The important values are now `probabilities.spam`, `probabilities.abusive`, and `probabilities.fine`; threshold them in code, and log the resolved response model so thresholds remain tied to the model build.
